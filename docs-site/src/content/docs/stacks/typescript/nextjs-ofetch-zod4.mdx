## When to use

You're on **Next.js 15 with the App Router** and want a single ValIO output
that drives both halves of the app:

- **Server components / route handlers / server actions** call generated
  `*.ofetch.ts` functions. `ofetch` plays nicely with Next.js's `fetch`
  cache, runs on both Node and Edge runtimes, and is tiny.
- **Client components** (`'use client'`) use generated `useGetPet`-style
  hooks backed by **Axios + TanStack Query 5** — interceptors for auth, an
  observable cache for optimistic updates, and per-component subscriptions.
- **Zod 4** parses the response in *both* paths, so the boundary between
  server and client is type-safe and runtime-validated.

The server prefetches and **dehydrates** the query cache; the client picks
it up via `<HydrationBoundary>` and never re-fetches the same data until
the cache goes stale.

## valio.config.json

Two transports, one validator, one query wrapper. ValIO emits a `.ofetch.ts`
**and** a `.axios.ts` file per operation, plus a `.query.ts` hook factory
that wraps Axios.

```json
{
  "projects": {
    "PetstoreApi": {
      "language": "typescript",
      "schema": "./openapi/petstore.yaml",
      "outputPath": "src/api/generated",
      "generateModels": true,
      "generateClients": true,
      "options": {
        "transports":   ["ofetch", "axios"],
        "validation":   "zod4",
        "queryWrapper": "tanstack-react",
        "enumFormat":   "union",
        "typescript": {
          "fetch": {
            "customFetcherPath": "../../lib/ofetchInstance",
            "customFetcherName": "apiOfetch"
          },
          "axios": {
            "customInstancePath": "../../lib/axiosInstance",
            "customInstanceName": "apiAxios"
          }
        }
      },
      "fileOrganization": {
        "separateModels":  true,
        "separateClients": true,
        "modelsPath":      "models",
        "clientsPath":     "api"
      }
    }
  }
}
```

The two `custom*` keys point ValIO at your hand-written instances so server
and client get their own `baseURL`, auth headers, retry policy, and timeouts
without touching generated code.

## What gets generated

| File                                  | Lives in           | Used by              |
|---------------------------------------|--------------------|----------------------|
| `models/<Model>.ts`                   | shared             | both halves          |
| `api/<op>/<op>.params.ts`             | shared             | both halves          |
| `api/<op>/<op>.ofetch.ts`             | RSC / route handler| server only          |
| `api/<op>/<op>.axios.ts`              | browser bundle     | client only          |
| `api/<op>/<op>.query.ts`              | browser bundle     | client only (hooks)  |
| `api/queryKeys.ts`                    | browser bundle     | client + server prefetch |
| `index.ts`                            | barrel             | both halves          |

The `*.ofetch.ts` transports are tree-shaken out of the client bundle by
Next.js (they only import from `ofetch`, which is server-friendly). The
`*.axios.ts` + `*.query.ts` pair is tree-shaken out of the server bundle
when no server file imports them.

    import { apiOfetch } from "../../../../lib/ofetchInstance";
    import { PetSchema, type Pet } from "../../models/Pet/Pet";
    import type { GetPetParams } from "./getPet.params";

    export interface GetPetOfetchOptions {
      /** Forwarded to Next.js fetch extensions: cache, revalidate, tags. */
      next?: { revalidate?: number | false; tags?: string[] };
      signal?: AbortSignal;
    }

    export async function getPet(
      params: GetPetParams,
      init: GetPetOfetchOptions = {},
    ): Promise<Pet> {
      const raw = await apiOfetch<unknown>(`/pets/${params.id}`, {
        method: "GET",
        signal: init.signal,
        // @ts-expect-error — next-only field, supported by ofetch in Next runtime
        next: init.next,
      });
      return PetSchema.parse(raw);
    }

    import { useQuery, type UseQueryOptions } from "@tanstack/react-query";
    import { getPet } from "./getPet.axios";
    import type { Pet } from "../../models/Pet/Pet";
    import type { GetPetParams } from "./getPet.params";
    import { petKeys } from "../../queryKeys";

    export type UseGetPetOptions = Omit<
      UseQueryOptions<Pet, Error, Pet, ReturnType<typeof petKeys.detail>>,
      "queryKey" | "queryFn"
    >;

    export function useGetPet(params: GetPetParams, options?: UseGetPetOptions) {
      return useQuery({
        queryKey: petKeys.detail(params.id),
        queryFn: ({ signal }) => getPet(params, { signal }),
        ...options,
      });
    }

     * transport so the result is identical to what the client hook would
     * resolve to after hydration.

    import {
      useMutation,
      useQueryClient,
      type UseMutationOptions,
    } from "@tanstack/react-query";
    import { createPet } from "./createPet.axios";
    import type { Pet } from "../../models/Pet/Pet";
    import type { CreatePetBody } from "./createPet.params";
    import { petKeys } from "../../queryKeys";

    export function useCreatePet(
      options?: UseMutationOptions<Pet, Error, CreatePetBody>,
    ) {
      const queryClient = useQueryClient();
      return useMutation({
        mutationFn: (body) => createPet(body),
        onSuccess: (created, variables, ctx) => {
          // Add to cache so the next render sees it without re-fetching.
          queryClient.setQueryData(petKeys.detail(created.id), created);
          queryClient.invalidateQueries({ queryKey: petKeys.lists() });
          options?.onSuccess?.(created, variables, ctx);
        },
        ...options,
      });
    }

    // src/lib/queryClient.ts — shared factory; server and client get their own instance.
    import {
      QueryClient,
      defaultShouldDehydrateQuery,
      isServer,
    } from "@tanstack/react-query";

    function make() {
      return new QueryClient({
        defaultOptions: {
          queries: { staleTime: 60_000 },
          dehydrate: {
            // Persist pending queries too so the client picks up where the
            // server left off (streaming RSC).
            shouldDehydrateQuery: (q) =>
              defaultShouldDehydrateQuery(q) || q.state.status === "pending",
          },
        },
      });
    }

    let clientSide: QueryClient | undefined;

    export function makeQueryClient() {
      if (isServer) return make();
      return (clientSide ??= make());
    }
    ```

    // src/app/pets/[id]/page.tsx — server component
    import { HydrationBoundary, dehydrate } from "@tanstack/react-query";
    import { makeQueryClient } from "@/lib/queryClient";
    import { getPet as fetchGetPet } from "@/api/generated/api/getPet/getPet.ofetch";
    import { petKeys } from "@/api/generated/queryKeys";
    import { PetDetail } from "./PetDetail";

    export default async function Page({ params }: { params: Promise<{ id: string }> }) {
      const { id } = await params;
      const numericId = Number(id);

      const queryClient = makeQueryClient();

      // Server-side prefetch with ofetch — feeds Next.js fetch cache + tags.
      await queryClient.prefetchQuery({
        queryKey: petKeys.detail(numericId),
        queryFn: ({ signal }) =>
          fetchGetPet({ id: numericId }, { next: { revalidate: 60, tags: ["pets"] }, signal }),
      });

      return (
        <HydrationBoundary state={dehydrate(queryClient)}>
          <PetDetail id={numericId} />
        </HydrationBoundary>
      );
    }
    ```

      );

## How the two halves cooperate

1. **Request arrives** → Next.js routes it to the server component.
2. **Server prefetches** via `*.ofetch.ts`. The response feeds Next.js's
   built-in `fetch` cache (with `revalidateTag('pets')` available for
   targeted invalidation from server actions).
3. **Server dehydrates** the `QueryClient` and ships the JSON snapshot
   inside `<HydrationBoundary>`.
4. **Client renders** — `Providers` mounts a single `QueryClient`,
   `HydrationBoundary` seeds it with the server snapshot, the
   `'use client'` component calls `useGetPet`/`useListPets`. **No
   refetch on initial mount** — TanStack sees fresh data already.
5. **Stale time elapses or focus event** → TanStack refetches *through
   Axios* (browser), keeping the auth interceptor, retry, and timeout
   policy you configured for the client side.
6. **Mutations** on the client (`useCreatePet`) hit Axios, update the
   cache optimistically, then call `revalidateTag('pets')` on the next
   navigation so the server's fetch cache stays in sync.

## Caching cheat sheet

| Layer                          | Server (RSC)                           | Client (browser)               |
|--------------------------------|----------------------------------------|--------------------------------|
| Transport                      | `ofetch` (powered by Node fetch)       | `axios`                        |
| Cache primitive                | Next.js `fetch` cache + `tags`         | TanStack `QueryClient`         |
| Invalidation API               | `revalidateTag('pets')` (server action)| `queryClient.invalidateQueries`|
| Background refresh             | Time-based `revalidate` directive      | `refetchOnWindowFocus`, intervals |
| Auth                           | `apiOfetch` interceptors (env tokens)  | `apiAxios` interceptors (cookies) |
| Validation                     | `PetSchema.parse(raw)`                 | `PetSchema.parse(res.data)`    |
| Runtime                        | Node / Edge                            | Browser                        |

## Server actions

Server actions (mutations from forms) call the same `*.ofetch.ts`
transport directly — no TanStack involvement on the server. Then
`revalidateTag('pets')` invalidates Next's cache; the client's next
mount picks up the new server snapshot.

```ts
// src/app/pets/actions.ts
"use server";

import { revalidateTag } from "next/cache";
import { createPet } from "@/api/generated/api/createPet/createPet.ofetch";

export async function addPetAction(formData: FormData) {
  const name = formData.get("name") as string;
  await createPet({ name });
  revalidateTag("pets");
}
```

## Streaming + Suspense (optional)

Drop the `await` on the prefetch and pass the in-flight `QueryClient`
straight into `<HydrationBoundary>`. TanStack 5 streams pending queries
to the client, so `<Suspense>` boundaries render their fallback until
the server pushes the resolved data over the same HTTP response.

## Tradeoffs

- **Bundle**: `ofetch` and `axios` ship in different graphs, so the
  client only loads `axios + @tanstack/react-query + zod` (~14 KB
  gzipped). Server doesn't bundle anything to the client from
  `*.ofetch.ts`.
- **Two-instance config**: you maintain two transport configs. The
  upside is per-runtime tuning (Node baseURL ≠ public baseURL, server
  uses bearer tokens, client uses cookies).
- **Same schemas**: Zod 4 schemas are shared, so a backwards-incompatible
  API change breaks the build in both places at once.
