import ky from "ky";
import type { Options } from "ky";
import { z } from "zod";
import type { GetUserByIdParams } from "./getUserById.params";
import type { GetUserByIdResult } from "./getUserById.results";
import { UserSchema } from "../../models/schemas/user/user.schema";

const GetUserByIdRouteParamsSchema = z.object({
    id: z.number().int()
});

/**
 * @description Get user by ID
 * @function
 * @async
 * @param {GetUserByIdParams} params
 * @param {Options} [options]
 * @param {AbortSignal} [signal]
 * @returns {Promise<GetUserByIdResult>}
 */
export async function getUserById(
    params: GetUserByIdParams,
    options?: Options,
    signal?: AbortSignal
): Promise<GetUserByIdResult> {
    const parsedRoute = GetUserByIdRouteParamsSchema.parse(params);
    
    const requestUrl = `/users/${parsedRoute.id}`;
    
    const kyOptions: Options = {
        ...(options ?? {}),
        throwHttpErrors: false,
        ...(signal ? {
        signal: signal
    } : {})
    };
    
    const response = await ky.get(
        requestUrl,
        kyOptions
    );
    const responseData = ((response.status === 204) ? undefined : await response.json<unknown>());
    
        if ((response.status === 200)) {
        return ({
            status: 200,
            body: UserSchema.parse(responseData),
            isSuccessful: true,
            isFailed: false
        } as GetUserByIdResult);
    }

    return ({
        status: response.status,
        body: responseData,
        isSuccessful: ((response.status >= 200) && (response.status < 300)),
        isFailed: !((response.status >= 200) && (response.status < 300))
    } as GetUserByIdResult);
}
